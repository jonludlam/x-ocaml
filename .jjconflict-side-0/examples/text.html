Text of all sorts

OCaml strings are bytes, so a string's length counts bytes, not characters. The output of each phrase appears under that phrase, whatever the text before it.

let camel = "🐫";; let bytes = String.length camel;; let chars = List.length (List.of_seq (String.to_seq camel)) let greetings = ["Grüß Gott"; "Καλημέρα"; "こんにちは"; "Привет"; "مرحبا"] let () = List.iter print_endline greetings (* A decomposed accent: e followed by a combining acute *) let e_acute = "é";; let n = String.length e_acute let decode s = let rec go i acc = if i >= String.length s then List.rev acc else let d = String.get_utf_8_uchar s i in go (i + Uchar.utf_decode_length d) (Uchar.utf_decode_uchar d :: acc) in go 0 [];; let code_points = List.map Uchar.to_int (decode "a🐫é")
