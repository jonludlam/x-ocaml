Code in a cell shows custom elements under it with X_ocaml_lib.display, and the elements call it back. They belong to the cell: running it again, or stopping the toplevel, takes them away.

A display, a callback, and an update from the callback

let count = ref 0 let shown = ref None let token = ref 0 let label () = Printf.sprintf {|{"label":"clicked %d","token":%d}|} !count !token let () = token := X_ocaml_lib.callback (fun _ -> incr count; Option.iter (fun d -> X_ocaml_lib.update d (label ())) !shown); shown := Some (X_ocaml_lib.display ~tag:"toy-display" (label ()))

An element defined after its display arrives

Its data, and an update sent before it is defined, wait for it; it gets the latest.

let d = X_ocaml_lib.display ~tag:"toy-late" {|{"label":"first","token":0}|} let () = X_ocaml_lib.update d {|{"label":"latest","token":0}|}

What a callback shows belongs to the callback's cell

let spawn = X_ocaml_lib.callback (fun _ -> ignore (X_ocaml_lib.display ~tag:"toy-display" {|{"label":"spawned","token":0}|})) let () = ignore (X_ocaml_lib.display ~tag:"toy-display" (Printf.sprintf {|{"label":"spawn","token":%d}|} spawn)) let bystander = ()

Running a cell again takes its displays and callbacks away

let notice = X_ocaml_lib.display ~tag:"toy-display" {|{"label":"nothing yet","token":0}|} let hits = ref 0 let () = let token = X_ocaml_lib.callback (fun _ -> incr hits) in ignore (X_ocaml_lib.display ~on_remove:(fun () -> X_ocaml_lib.update notice {|{"label":"on_remove ran","token":0}|}) ~tag:"toy-display" (Printf.sprintf {|{"label":"rerun","token":%d}|} token))

A library's element, loaded by require

The script is relative to the worker, and is loaded once however often it is asked for.

let () = X_ocaml_lib.require ~tag:"toy-required" ~src:"toy-element.js"; X_ocaml_lib.require ~tag:"toy-required" ~src:"toy-element.js"; ignore (X_ocaml_lib.display ~tag:"toy-required" "hello")

An HTML display, updated while its cell still runs

let bar = X_ocaml_lib.html "starting" let () = for i = 1 to 5 do let t = Sys.time () in while Sys.time () -. t < 0.3 do () done; X_ocaml_lib.update bar (Printf.sprintf "<b data-step=\"%d\">step %d of 5</b>" i i) done

Controls in an HTML display call back

let said = X_ocaml_lib.html "nothing" let say s = X_ocaml_lib.update said s let token = X_ocaml_lib.callback say let () = ignore (X_ocaml_lib.html (Printf.sprintf {|<button data-callback="%d" data-payload="pressed">press</button> <input type="range" data-callback="%d" value="10"> <input type="checkbox" data-callback="%d"> <span data-callback="%d">no payload</span>|} token token token token))

Removing a display, silly tags and tokens, a callback that raises

let gone = X_ocaml_lib.display ~tag:"toy-display" {|{"label":"going","token":0}|} let () = X_ocaml_lib.remove gone let () = ignore (X_ocaml_lib.display ~tag:"not a tag" "x") let raises = X_ocaml_lib.callback (fun _ -> failwith "on purpose") let () = ignore (X_ocaml_lib.display ~tag:"toy-display" (Printf.sprintf {|{"label":"raise","token":%d}|} raises)) let still = "answering" let hits_now = !hits
