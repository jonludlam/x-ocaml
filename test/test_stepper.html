A cell records a computation's states with Focs_check.Steps, and an <x-stepper> shows them one at a time, as box-and-pointer diagrams. In predict mode the reader picks the next state before it is shown; in lecture mode a script steps it, as a slide's exec block would.

rev with an accumulator, to predict

open Focs_check let show l = "[" ^ String.concat "; " (List.map string_of_int l) ^ "]" let rev_steps l = let t = Steps.create () in let st ?note label xs acc = Steps.state ?note ~label ~highlight:[ Steps.list string_of_int acc ] [ ("l", Steps.list string_of_int l); ("xs", Steps.list string_of_int xs); ("acc", Steps.list string_of_int acc) ] in let go_label xs acc = Printf.sprintf "go %s %s" (show xs) (show acc) in Steps.add t (st ("rev " ^ show l ^ " = " ^ go_label l []) l []); let rec go acc xs = match xs with | [] -> acc | x :: rest -> let acc' = x :: acc in let wrong = [ ( Printf.sprintf "x :: acc puts %d in front of the accumulator, not after it: \ the accumulator ends up reversed." x, acc @ [ x ] ); ("Nothing is lost: the head of the list moves onto the accumulator.", acc); ("The accumulator is extended, not replaced: it keeps what it had.", [ x ]) ] |> List.filter (fun (_, a) -> a <> acc') |> List.sort_uniq (fun (_, a) (_, b) -> compare a b) |> List.map (fun (why, a) -> (why, st (go_label rest a) rest a)) in Steps.add t ~choices:wrong (st (go_label rest acc') rest acc'); go acc' rest in let r = go [] l in Steps.add t ~choices:[ ("go returns the accumulator, which holds the elements in the \ reverse of the order they were met.", Steps.state ~label:("= " ^ show l) [ ("result", Steps.list string_of_int l) ]) ] (Steps.state ~label:("= " ^ show r) ~note:"go [] acc = acc" [ ("result", Steps.list string_of_int r) ]); t let () = Steps.show ~id:"rev-steps" ~title:"rev, with an accumulator" (rev_steps [ 1; 2; 3 ])

A preorder traversal, in lecture mode

open Focs_check type 'a tree = Lf | Br of 'a * 'a tree * 'a tree let view = function Lf -> None | Br (v, l, r) -> Some (v, [ l; r ]) let preorder_steps t = let s = Steps.create () in let tree = Steps.tree string_of_int view t in let out = ref [] in let st label here = Steps.state ~label ~highlight:[ Steps.tree string_of_int view here ] [ ("t", tree); ("here", Steps.tree string_of_int view here); ("visited", Steps.list string_of_int (List.rev !out)) ] in Steps.add s (st "preorder t" t); let rec go = function | Lf -> () | Br (v, l, r) as here -> out := v :: !out; Steps.add s (st (Printf.sprintf "visit %d" v) here); go l; go r in go t; s let t = Br (4, Br (2, Br (1, Lf, Lf), Br (3, Lf, Lf)), Br (6, Br (5, Lf, Lf), Lf)) let () = Steps.show ~id:"pre-steps" ~mode:"lecture" ~title:"preorder" (preorder_steps t)
