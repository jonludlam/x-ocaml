OCaml 5's bytecode compiler starts code whose frame needs more than about 32 words of stack with a call to caml_ensure_stack_capacity, which the toplevel must know. A module holds its definitions on the stack until it is built, so a dozen or so of them is enough.

module Units = struct let km_of_miles m = m *. 1.609344 let miles_of_km k = k /. 1.609344 let m_of_feet f = f *. 0.3048 let feet_of_m m = m /. 0.3048 let cm_of_inches i = i *. 2.54 let inches_of_cm c = c /. 2.54 let kg_of_pounds p = p *. 0.45359237 let pounds_of_kg k = k /. 0.45359237 let g_of_ounces o = o *. 28.349523125 let ounces_of_g g = g /. 28.349523125 let l_of_pints p = p *. 0.56826125 let pints_of_l l = l /. 0.56826125 let c_of_f f = (f -. 32.) *. 5. /. 9. let f_of_c c = c *. 9. /. 5. +. 32. let kmh_of_mph m = km_of_miles m let mph_of_kmh k = miles_of_km k end let marathon = Units.km_of_miles 26.2
