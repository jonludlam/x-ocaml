The toolkit in ui/, against real cells: what variables do, what the controls set, and the awkward cases: functions as choices, typing fast, a slow clock, a handler that raises, a cell run again.

A handler's changes are drawn once

open Ui let a = Var.create 0 and b = Var.create 0 let renders = ref 0 let () = show ~id:"batch" (vbox [ button "both" (fun () -> Var.update a succ; Var.update b succ); dyn (fun () -> incr renders; element "span" ~attrs:[ ("data-t", "batch") ] [ text (Printf.sprintf "%d %d %d" (Var.get a) (Var.get b) !renders) ]) ])

link and dlink

let x = Var.create 1 and y = Var.create 2 let () = link x y let doubled = Var.create 0 let () = dlink ~f:(fun v -> 2 * v) x doubled let () = show ~id:"links" (vbox [ button "x+1" (fun () -> Var.update x succ); button "y+10" (fun () -> Var.update y (( + ) 10)); dyn (fun () -> element "span" ~attrs:[ ("data-t", "links") ] [ text (Printf.sprintf "%d %d %d" (Var.get x) (Var.get y) (Var.get doubled)) ]) ])

Choices whose values are functions

Functions can only be compared by identity, and each mention of a primitive such as sin makes a new closure: the controls share one list of options.

let options = [ ("sin", sin); ("cos", cos); ("exp", exp) ] let f = Var.create (List.assoc "sin" options) let () = show ~id:"funs" (vbox [ dropdown options f; toggle_buttons (List.filteri (fun i _ -> i < 2) options) f; dyn (fun () -> element "span" ~attrs:[ ("data-t", "funs") ] [ text (Printf.sprintf "%g" ((Var.get f) 0.)) ]) ])

Typing into a text input whose variable redraws the page

let name = Var.create "" let () = show ~id:"typing" (vbox [ text_input ~placeholder:"type" name; dyn (fun () -> element "span" ~attrs:[ ("data-t", "typed") ] [ text (String.uppercase_ascii (Var.get name)) ]) ])

Picking several, tags, and a range

let picked = Var.create [] let words = Var.create [ "one" ] let band = Var.create (20, 60) let () = show ~id:"multi" (vbox [ select_multiple (List.map (fun s -> (s, s)) [ "a"; "b"; "c" ]) picked; tags words; range_slider band; dyn (fun () -> let lo, hi = Var.get band in element "span" ~attrs:[ ("data-t", "multi") ] [ text (Printf.sprintf "%s|%s|%d-%d" (String.concat "," (Var.get picked)) (String.concat "," (Var.get words)) lo hi) ]) ])

A table: numbers sort as numbers, filtering, pages

let () = show ~id:"table" (table ~page_size:2 ~header:[ "name"; "n" ] (fun () -> [ [ "ten"; "10" ]; [ "nine"; "9" ]; [ "hundred"; "100" ] ]))

Tabs keep what their controls set

let level = Var.create 3 let () = show ~id:"tabs" (tabs [ ("one", int_slider ~max:10 level); ("two", text_of (Printf.sprintf "level %d") level) ])

A slow clock does not pile its ticks up

Each tick takes 100 ms and the clock asks for one every 16 ms. The ticks keep pace with the worker, and none are left over once it stops.

let ticks = Var.create 0 and ticking = Var.create false let () = show ~id:"clock" (vbox [ button "go" (fun () -> Var.set ticking true); button "halt" (fun () -> Var.set ticking false); element "span" ~attrs:[ ("data-t", "ticks") ] [ text_of string_of_int ticks ]; clock ~interval:16 ~running:ticking (fun () -> let t = Sys.time () in while Sys.time () -. t < 0.1 do () done; Var.update ticks succ) ])

A click on a chart, in the chart's coordinates

let clicked = Var.create (0., 0.) let () = show ~id:"chart" (vbox [ Plot.chart ~x_range:(0., 10.) ~y_range:(0., 10.) ~on_click:(fun p -> Var.set clicked (p.Plot.x, p.Plot.y)) [ Plot.scatter [ (5., 5.) ] ]; dyn (fun () -> let x, y = Var.get clicked in element "span" ~attrs:[ ("data-t", "clicked") ] [ text (Printf.sprintf "%.1f,%.1f" x y) ]) ])

A handler that raises, and a widget that cannot be drawn

let broken = Var.create false let () = show ~id:"raise" (vbox [ button "raise" (fun () -> failwith "on purpose"); button "break" (fun () -> Var.set broken true); dyn (fun () -> if Var.get broken then failwith "cannot draw" else text "drawn") ])

Keys, and interact's manual mode

let last = Var.create "none" let () = show ~id:"keys" (vbox [ keyboard ~on_key:(fun e -> Var.set last e.Event.key) (text "focus me"); element "span" ~attrs:[ ("data-t", "key") ] [ text_of Fun.id last ]; interact ~manual:true (let+ n = Interact.int "n" ~max:10 2 in element "span" ~attrs:[ ("data-t", "manual") ] [ text (string_of_int (n * n)) ]) ])

A cell run again leaves no ghost behind

The second cell shows a variable of the first. Run it again: its old widget goes, and when the variable changes, only the new one is drawn.

let shared = Var.create 0 let () = show ~id:"source" (button "bump" (fun () -> Var.update shared succ)) let () = show (element "span" ~attrs:[ ("data-t", "viewer") ] [ text_of string_of_int shared ])
