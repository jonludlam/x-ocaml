A test cell reads the code of the exercise it is for, by its data-id, and loads it apart from the cells, as a module of its own, with what the compiler said about it (Focs_check). The page runs on the plain worker, or on the effects worker with ?worker=effects.

A test reads its exercise's code

let rec rev = function [] -> [] | x :: xs -> rev xs @ [ x ] let () = print_endline ("read: " ^ Focs_check.source "rev-ex")

It loads it as a module, which its later phrases use

let s = Focs_check.load ~as_:"Student" "rev-ex" let () = assert s.ok ;; let () = assert (Student.rev [ 1; 2; 3 ] = [ 3; 2; 1 ]) let () = print_endline "Student.rev works"

The module stays for later cells

let later = Student.rev [ 'a'; 'b' ]

A type error is a diagnostic, at the page's line

let ok = 1 let wrong : int = "two" let s = Focs_check.load ~as_:"Bad" "bad-ex" let () = Focs_check.print_diagnostics s; Printf.printf "ok: %b\n" s.ok; match s.diagnostics with | [ d ] -> Printf.printf "error at line %d\n" d.line | _ -> print_endline "not one diagnostic"

Warnings are kept, with the flags asked for

let rec nsum n = if n = 0 then 0 else n + (nsum [@tailcall]) (n - 1) let s = Focs_check.load ~warnings:"@51" ~as_:"Tail" "tail-ex" let () = Focs_check.print_diagnostics s; List.iter (fun (d : Focs_check.diagnostic) -> if d.warning = Some 51 then Printf.printf "warning 51 at line %d, error: %b\n" d.line d.error) s.diagnostics

Code that raises as it loads

let x = List.hd [] let s = Focs_check.load ~as_:"Raises" "raise-ex" let () = Focs_check.print_diagnostics s; Printf.printf "ok: %b\n" s.ok

Loading leaves the cells' own definitions alone

let shared = "student" let shared = "test" let s = Focs_check.load ~as_:"Iso" "iso-ex" ;; let () = Printf.printf "shared: %s, Iso.shared: %s\n" shared Iso.shared

An edited exercise is read afresh

let v = 1 let () = print_endline ("now: " ^ Focs_check.source "edit-ex")
