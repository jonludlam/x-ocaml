Tests that check more than answers, with Focs_check: whether the exercise is tail recursive, how its work grows with its input, and what it uses. The page runs on the plain worker, or on the effects worker with ?worker=effects.

nsum is not tail recursive, and its test says where

let rec nsum n = if n = 0 then 0 else n + nsum (n - 1) let t = Focs_check.tailcheck "nsum" let () = Focs_check.print_tail t let () = assert (Focs_check.is_tail_recursive t)

summing is

let rec summing n total = if n = 0 then total else summing (n - 1) (n + total) let t = Focs_check.tailcheck "summing" let () = Focs_check.print_tail t let () = assert (Focs_check.is_tail_recursive t)

nrev's work grows as the square of its input

let rec nrev = function [] -> [] | x :: xs -> nrev xs @ [ x ] let s = Focs_check.load_counting ~as_:"Nrev" "nrev" let g = Focs_check.growth (Focs_check.measure ~as_:"Nrev" ~input:"fun n -> List.init n Fun.id" ~call:"Nrev.nrev" [ 64; 128; 256; 512; 1024 ]) let () = Focs_check.print_growth g let () = assert (g.model = "O(n)")

rev with an accumulator, linearly

let rev l = let rec go acc = function [] -> acc | x :: xs -> go (x :: acc) xs in go [] l let s = Focs_check.load_counting ~as_:"Rev" "rev" let g = Focs_check.growth (Focs_check.measure ~as_:"Rev" ~input:"fun n -> List.init n Fun.id" ~call:"Rev.rev" [ 64; 128; 256; 512; 1024 ]) let () = Focs_check.print_growth g let () = assert (g.model = "O(n)")

List.rev is not allowed, even under another name

module L = List let rev l = L.rev l let broken = Focs_check.check_uses [ Forbid_value "Stdlib.List.rev" ] "sneaky" let () = List.iter (fun (line, msg) -> Printf.printf "Line %d: %s\n" line msg) broken let () = assert (broken = [])

A loop that never ends runs out of fuel

let rec spin l = spin l let s = Focs_check.load_counting ~as_:"Spin" "loop" let r = Focs_check.measure ~fuel:100_000 ~as_:"Spin" ~input:"Fun.id" ~call:"Spin.spin" [ 1 ] let () = print_endline (if r = [ (1, None) ] then "out of fuel" else "?")
