Cells draw widgets with js_top_worker's Widget library, and the reader's clicks and typing go back to the handlers in the worker. Every update patches the widget where it is.

A counter: clicks go to the worker and the view is patched in place

open Widget.View let count = ref 0 let view () = Element { tag = "div"; attrs = []; children = [ Element { tag = "button"; attrs = [ Property ("data-t", "inc"); Handler ("click", "inc") ]; children = [ Text "+1" ] }; Element { tag = "span"; attrs = [ Property ("data-t", "count") ]; children = [ Text (string_of_int !count) ] } ] } let () = Widget.display ~id:"counter" ~handlers:[ ("inc", fun _ -> incr count; Widget.update ~id:"counter" (view ())) ] (view ())

An input that echoes: typing keeps its focus and its keystrokes

The program writes what it was sent back into the input's value, a keystroke behind the reader. Clearing it from the program still works.

open Widget.View let typed = ref "" let echo () = Element { tag = "div"; attrs = []; children = [ Element { tag = "input"; attrs = [ Property ("data-t", "field"); Property ("value", !typed); Handler ("input", "typed") ]; children = [] }; Element { tag = "button"; attrs = [ Property ("data-t", "wipe"); Handler ("click", "wipe") ]; children = [ Text "clear" ] }; Element { tag = "p"; attrs = [ Property ("data-t", "said") ]; children = [ Text ("you typed: " ^ !typed) ] } ] } let () = Widget.display ~id:"echo" ~handlers:[ ("typed", fun v -> typed := Option.value v ~default:""; Widget.update ~id:"echo" (echo ())); ("wipe", fun _ -> typed := ""; Widget.update ~id:"echo" (echo ())) ] (echo ()) (* The same, answering slowly, so that echoes arrive while the reader types. *) let slow = ref "" let slow_echo () = Element { tag = "div"; attrs = []; children = [ Element { tag = "input"; attrs = [ Property ("data-t", "slow"); Property ("value", !slow); Handler ("input", "typed") ]; children = [] }; Element { tag = "p"; attrs = [ Property ("data-t", "slow-said") ]; children = [ Text !slow ] } ] } let () = Widget.display ~id:"slow-echo" ~handlers:[ ("typed", fun v -> let t = Sys.time () in while Sys.time () -. t < 0.06 do () done; slow := Option.value v ~default:""; Widget.update ~id:"slow-echo" (slow_echo ())) ] (slow_echo ())

The view decides what an input shows

An input whose view always says "" is emptied after each entry; a program that rewrites what is typed has the last word; a checked box follows the program even when clicked faster than it answers.

open Widget.View let entries = ref [] and shout = ref "" and on = ref false let view () = Element { tag = "div"; attrs = []; children = [ Element { tag = "input"; attrs = [ Property ("data-t", "entry"); Property ("value", ""); Handler ("change", "add") ]; children = [] }; Element { tag = "span"; attrs = [ Property ("data-t", "entries") ]; children = [ Text (String.concat "," (List.rev !entries)) ] }; Element { tag = "input"; attrs = [ Property ("data-t", "shout"); Property ("value", !shout); Handler ("input", "shout") ]; children = [] }; Element { tag = "input"; attrs = [ Property ("type", "checkbox"); Property ("data-t", "flip"); Handler ("click", "flip") ] @ (if !on then [ Property ("checked", "") ] else []); children = [] }; Element { tag = "span"; attrs = [ Property ("data-t", "on") ]; children = [ Text (string_of_bool !on) ] } ] } let () = let redraw () = Widget.update ~id:"controlled" (view ()) in Widget.display ~id:"controlled" ~handlers:[ ("add", fun v -> entries := Option.value v ~default:"?" :: !entries; redraw ()); ("shout", fun v -> shout := String.uppercase_ascii (Option.value v ~default:""); redraw ()); ("flip", fun v -> on := v = Some "true"; redraw ()) ] (view ())
