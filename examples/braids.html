Braids

Words of crossings, and the pictures that make sense of them.

Take a few strands side by side, hanging from a bar, and cross neighbouring ones over each other as you go down: that is a braid. Write down only the crossings, in order from the top, and you have a word. Number the places 1, 2, 3, … from the left: σi says that the strand in place i crosses over its neighbour in place i+1, and σi−1 that it passes under it. So [1; 2; -1; 3] is σ1σ2σ1−1σ3 on four strands.

As a list of numbers that says very little. Drawn, it says a lot. The cells on this page are OCaml, run in your browser: the first few define braids and draw them, and from then on every braid a cell computes is shown as a picture under the line that computed it. You can change any cell and run it again.

A small braid library

A braid is its number of strands and its word. sigma i is one crossing, a ** b is a with b hung below it, and inverse b undoes b.

(* A braid on [strands] strands is a word of crossings, read from the top down: [i] is sigma_i, where the strand in place i crosses over the strand in place i+1, and [-i] is its inverse, where it passes under. *) type braid = { strands : int; word : int list } let sigma i = { strands = i + 1; word = [ i ] } let identity n = { strands = n; word = [] } (* [a ** b] is a, then b below it. *) let ( ** ) a b = { strands = max a.strands b.strands; word = a.word @ b.word } (* Undo the crossings, last first: the mirror image, top to bottom. *) let inverse b = { b with word = List.rev_map (fun c -> -c) b.word } let rec power b k = if k = 0 then identity b.strands else if k < 0 then power (inverse b) (-k) else b ** power b (k - 1)

To draw a braid, we follow its strands down. Each strand is named by the place it starts in, and a crossing swaps the two strands it involves. At the bottom, each strand has ended up somewhere: that is the braid's permutation.

(* Follow the strands down. Name each strand by the place it starts in; [places] lists the strand in each place. A crossing [i] or [-i] swaps the strands in places i and i+1. *) let start n = List.init n (fun p -> p + 1) let swap c places = let i = abs c in List.mapi (fun p s -> if p = i - 1 then List.nth places i else if p = i then List.nth places (i - 1) else s) places let finish b = List.fold_left (fun ps c -> swap c ps) (start b.strands) b.word (* [permutation b] lists, for strands 1, 2, ..., n, the place each ends in. *) let rec place_of s = function | [] -> 0 | t :: ts -> if t = s then 1 else 1 + place_of s ts let permutation b = List.map (fun s -> place_of s (finish b)) (start b.strands)

The rest is drawing. Each crossing is one row: two curves, and the one that passes under is cut short around the middle, so that you see it go behind. Each strand keeps the colour of the place it started in, marked by a dot at the top.

The drawing code: text, curves, and SVG (* The braid as text: runs of one crossing are written as powers, so [[1; 1; 1; -2]] is sigma_1 cubed, then sigma_2 inverse. The glyphs are given by their Unicode code points, to keep this cell ASCII. *) let glyph code = let b = Buffer.create 4 in Buffer.add_utf_8_uchar b (Uchar.of_int code); Buffer.contents b let digits glyph_of n = String.to_seq (string_of_int n) |> Seq.map (fun d -> glyph (glyph_of (Char.code d - Char.code '0'))) |> List.of_seq |> String.concat "" (* Subscript digits are U+2080 to U+2089; superscript 1, 2 and 3 are older, at U+00B9, U+00B2 and U+00B3. *) let sub = digits (fun d -> 0x2080 + d) let sup = digits (fun d -> if d = 1 then 0xB9 else if d = 2 || d = 3 then 0xB0 + d else 0x2070 + d) let sigma_glyph = glyph 0x3C3 and minus = glyph 0x207B let letter (c, k) = let e = if c > 0 then k else -k in let power = if e = 1 then "" else if e > 0 then sup e else minus ^ sup (-e) in sigma_glyph ^ sub (abs c) ^ power let rec runs = function | [] -> [] | c :: rest -> ( match runs rest with | (c', k) :: more when c' = c -> (c, k + 1) :: more | more -> (c, 1) :: more) let word_string b = String.concat "" (List.map letter (runs b.word)) let to_string b = if b.word = [] then Printf.sprintf "no crossings, %d strands" b.strands else Printf.sprintf "%s, %d strands" (word_string b) b.strands (* The drawing: strand places are [gap] apart, each crossing a row [row] tall. A crossing is two cubic curves; the one passing under is cut short around the middle, so the gap shows whatever is behind. *) let gap = 30. let row = 32. let margin = 12. let x_of place = margin +. (gap *. float (place - 1)) let colours = [ "#e15759"; "#4e79a7"; "#59a14f"; "#f28e2b"; "#b07aa1"; "#76b7b2"; "#edc948"; "#9c755f" ] let colour s = List.nth colours ((s - 1) mod List.length colours) let pt (x, y) = Printf.sprintf "%.1f %.1f" x y let line s (x0, y0) (x1, y1) = Printf.sprintf "<path d='M%s L%s' stroke='%s'/>" (pt (x0, y0)) (pt (x1, y1)) (colour s) let cubic s (p0, p1, p2, p3) = Printf.sprintf "<path d='M%s C%s %s %s' stroke='%s'/>" (pt p0) (pt p1) (pt p2) (pt p3) (colour s) (* A strand going from place [a] to place [b] in the row at height [y]. *) let s_curve a b y = let x0 = x_of a and x1 = x_of b and m = y +. (row /. 2.) in ((x0, y), (x0, m), (x1, m), (x1, y +. row)) (* The part of a cubic from [t] on, and up to [t]: de Casteljau. *) let mix (x0, y0) (x1, y1) t = (x0 +. (t *. (x1 -. x0)), y0 +. (t *. (y1 -. y0))) let split (p0, p1, p2, p3) t = let a = mix p0 p1 t and b = mix p1 p2 t and c = mix p2 p3 t in let d = mix a b t and e = mix b c t in let m = mix d e t in ((p0, a, d, m), (m, e, c, p3)) (* The under strand, with a gap between [0.36] and [0.64] of the way. *) let under s q = let first, rest = split q 0.36 in let _, last = split rest (0.28 /. 0.64) in cubic s first ^ cubic s last (* One row: the crossing [c] and, beside it, straight strands. *) let straight y p s = line s (x_of p, y) (x_of p, y +. row) let crossing y c places = let i = abs c in let a = List.nth places (i - 1) and b = List.nth places i in let right = s_curve i (i + 1) y and left = s_curve (i + 1) i y in if c > 0 then under b left ^ cubic a right else under a right ^ cubic b left let draw_row y c places = let others = List.mapi (fun p s -> if p + 1 = abs c || p = abs c then "" else straight y (p + 1) s) places in String.concat "" others ^ crossing y c places (* All the rows from height [y] down; returns the drawing and the strands' places at the bottom. *) let rec draw_rows y places = function | [] -> ("", places) | c :: rest -> let drawn, bottom = draw_rows (y +. row) (swap c places) rest in (draw_row y c places ^ drawn, bottom)

Last, pp is a printer for braids: it gives the page the drawing, with X_ocaml_lib.output_html, and the toplevel a line of text. #install_printer tells the toplevel to use it for every braid it shows.

let svg ~width ~height body = Printf.sprintf "<svg xmlns='http://www.w3.org/2000/svg' width='%.0f' height='%.0f' viewBox='0 0 %.0f %.0f' fill='none' stroke-width='4.5' stroke-linecap='round'>%s</svg>" width height width height body let dot s (x, y) = Printf.sprintf "<circle cx='%.1f' cy='%.1f' r='4.5' fill='%s'/>" x y (colour s) (* The braid, with a dot where each strand starts, and a short lead above and below (a longer one if there are no crossings). *) let lead b = if b.word = [] then row else 6. let draw b = let top = margin +. 6. and n = b.strands in let rows, bottom = draw_rows top (start n) b.word in let y = top +. (row *. float (List.length b.word)) in let tail p s = line s (x_of (p + 1), y) (x_of (p + 1), y +. lead b) in let head s = line s (x_of s, margin) (x_of s, top) ^ dot s (x_of s, margin) in let body = String.concat "" (List.map head (start n)) ^ rows ^ String.concat "" (List.mapi tail bottom) in svg ~width:(x_of n +. margin) ~height:(y +. lead b +. margin) body let pp ppf b = X_ocaml_lib.output_html (draw b); Format.pp_print_string ppf (to_string b) #install_printer pp

One crossing, then more

Here is σ1: the red strand crosses over the blue one. Its inverse is the same crossing the other way: red goes under.

let over = sigma 1 let under = inverse (sigma 1)

Composing braids hangs one below the other. The inverse of a braid undoes its crossings, last first: it is its mirror image, top to bottom.

let b = sigma 1 ** sigma 2 ** inverse (sigma 1) ** sigma 3 let b' = inverse b

Pulling it apart

A braid followed by its inverse looks like a tangle. But each crossing in the middle meets its own undoing: pull the strands tight and they come apart, one pair of crossings after another, until they hang straight. Cancelling σiσi−1 wherever they meet is free reduction, and it does the same to the word.

let tangle = b ** inverse b (* Free reduction: a crossing followed at once by its inverse cancels. The stack holds what is left so far, last crossing on top. *) let reduce b = let step stack c = match stack with d :: rest when d = -c -> rest | _ -> c :: stack in { b with word = List.rev (List.fold_left step [] b.word) } let pulled = reduce tangle

Two words, one braid

Different words can be the same braid. Follow each strand down in these two: red goes from the left to the right, green from the right to the left, blue stays in the middle, and in both, red passes over blue and green, and blue over green. Slide the middle crossing of one to the side and you have the other. This is the braid relation, σ1σ2σ1 = σ2σ1σ2.

let left = sigma 1 ** sigma 2 ** sigma 1 let right = sigma 2 ** sigma 1 ** sigma 2

Crossings far apart, which share no strand, slide past each other: σ1σ3 = σ3σ1. With cancelling a crossing against its inverse, and putting such a pair in, these two rules are all there is: two words are the same braid when one can be turned into the other with them (Artin, 1925).

let first = sigma 1 ** sigma 3 let second = sigma 3 ** sigma 1

But turning one word into another by hand is a search, and it may not be clear where to start. Artin found a test that needs none. Think of the strands' lower ends as pegs in a board, and of a braid as a way of moving the pegs around each other: follow the strands down, and each crossing swaps two neighbouring pegs, one passing in front of the other.

Now put elastic bands on the board, one for each peg: from a nail at the edge, around the peg, and back to the nail. Move the pegs as the braid says, and the bands are dragged along, never cut and never lifted over a peg, so a band that went around one peg can end up winding around several. Write each band down as the pegs it goes around, in order, with a minus for going around one the other way. Two braids are the same exactly when they leave every band the same, once a peg gone around and straight back is crossed out. The cell below does just that: loops writes down where a braid leaves each band, and equal compares them.

(* Artin's action: a braid moves loops around the strands' feet, and each loop is a word in the free group on x_1 ... x_n, here [j] for x_j and [-j] for its inverse. Two braids are equal exactly when they move every x_j to the same loop (Artin, 1925). *) let rec cancel = function | [] -> [] | g :: rest -> ( match cancel rest with h :: more when h = -g -> more | more -> g :: more) let invert w = List.rev_map (fun g -> -g) w (* Where the crossing [c], [i] or [-i], sends the loop x_j. *) let image c j = let i = abs c in if c > 0 then (if j = i then [ i; i + 1; -i ] else if j = i + 1 then [ i ] else [ j ]) else if j = i then [ i + 1 ] else if j = i + 1 then [ -(i + 1); i; i + 1 ] else [ j ] let act c w = cancel (List.concat_map (fun g -> if g > 0 then image c g else invert (image c (-g))) w) let loops b = List.map (fun j -> List.fold_left (fun w c -> act c w) [ j ] b.word) (start b.strands) let equal a b = let n = max a.strands b.strands in loops { a with strands = n } = loops { b with strands = n } equal left right;; equal first second;; equal (sigma 1 ** sigma 2) (sigma 2 ** sigma 1);; equal tangle (identity 4);; equal (sigma 1 ** sigma 2 ** inverse (sigma 1)) (inverse (sigma 2) ** sigma 1 ** sigma 2)

The last pair has no crossing to cancel, and neither word is the other with crossings slid about, yet they are the same braid. Draw them and look.

let lhs = sigma 1 ** sigma 2 ** inverse (sigma 1) let rhs = inverse (sigma 2) ** sigma 1 ** sigma 2

Where the strands go

The colours let you follow a strand from its dot to the bottom. A plait, the way hair is braided, takes the left strand over the middle, then the right strand over the new middle, again and again: (σ1σ2−1)6.

let plait = power (sigma 1 ** inverse (sigma 2)) 6

Every strand visits every place, and after three plaits each is back where it began: the permutation of one plait sends strand 1 to place 3, 2 to 1 and 3 to 2.

permutation (sigma 1 ** inverse (sigma 2));; permutation plait

Half a turn, and a whole one

Hold the top of the strands still and turn the bottom through half a turn, as you would a ribbon: every pair of strands crosses once. That is Garside's half twist, Δ. It reverses the strands.

(* Garside's half twist, Delta: turn the bottom of the strands through half a turn, like a ribbon. [climb i] is [1; 2; ...; i]; Delta on n strands climbs to n-1, then to n-2, ..., then to 1. *) let rec climb i = if i = 0 then identity 1 else climb (i - 1) ** sigma i let rec delta n = if n <= 1 then identity n else climb (n - 1) ** delta (n - 1) let full_twist n = power (delta n) 2 let half = delta 4 permutation half

Turn it another half turn, and every strand is back in its own place. The full twist Δ2 permutes nothing, yet it is not the braid of straight strands: every strand wraps once around all the others, and no pulling undoes that.

let full = full_twist 4 permutation full;; equal full (identity 4)

The half twist turns the braid over: a crossing slid down through it comes out in the mirror place, so Δσ1 = σ3Δ. The full twist turns it over twice, and every braid slides through it untouched: it commutes with everything.

equal (half ** sigma 1) (sigma 3 ** half);; equal (full ** b) (b ** full);; equal (half ** b) (b ** half)

Tie the ends

Join the bottom of each strand to the top of its place, round the side, and a braid becomes a knot, or a link of several loops, one for each cycle of its permutation. Every knot is the closure of some braid (Alexander, 1923). Each loop is drawn in one colour.

(* Close a braid: join the bottom of each strand back round to the top of its place. The result is a knot, or a link of several: one loop for each cycle of the permutation. Colour each loop by its smallest strand. *) type closure = Closure of braid let rec orbit b s p = if p = s then [] else p :: orbit b s (List.nth (permutation b) (p - 1)) let loop_of b s = List.fold_left min s (orbit b s (List.nth (permutation b) (s - 1))) let components b = List.length (List.filter (fun s -> loop_of b s = s) (start b.strands)) (* The loop from place [p] at the bottom [y1] back to the top [y0]: a U round to the right, up, and back over. *) let back n y0 y1 p = let x = x_of p and x' = x_of n +. (gap *. float (n - p + 1)) in let k = 0.55 *. (x' -. x) in Printf.sprintf "M%s C%s %s %s L%s C%s %s %s" (pt (x, y1)) (pt (x, y1 +. k)) (pt (x', y1 +. k)) (pt (x', y1)) (pt (x', y0)) (pt (x', y0 -. k)) (pt (x, y0 -. k)) (pt (x, y0)) let draw_closure b = let n = b.strands in let k = (0.75 *. 0.55 *. gap *. float (2 * n - 1)) +. margin in let rows, bottom = draw_rows k (List.map (loop_of b) (start n)) b.word in let y1 = k +. (row *. float (List.length b.word)) in let arc p s = Printf.sprintf "<path d='%s' stroke='%s'/>" (back n k y1 p) (colour s) in let arcs = List.mapi (fun p s -> arc (p + 1) s) bottom in svg ~width:(x_of n +. (gap *. float n) +. margin) ~height:(y1 +. k) (rows ^ String.concat "" arcs) let pp_closure ppf (Closure b) = X_ocaml_lib.output_html (draw_closure b); let c = components b in Format.fprintf ppf "the closure of %s: %s" (word_string b) (if c = 1 then "a knot" else Printf.sprintf "a link of %d loops" c) #install_printer pp_closure

Three crossings of two strands close up into the trefoil, the simplest true knot; two into the Hopf link, two rings through each other. The plait's first two rounds give the figure eight knot.

let trefoil = Closure (power (sigma 1) 3) let hopf = Closure (power (sigma 1) 2) let figure_eight = Closure (power (sigma 1 ** inverse (sigma 2)) 2)
