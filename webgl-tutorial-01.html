WebGL Tutorial 1: Getting Started with WebGL

This tutorial introduces WebGL basics. We'll initialize a WebGL context and clear the canvas to a solid color. This is the foundation for all WebGL applications.

Example: Clear to Black

The simplest WebGL program: get a WebGL context and clear the canvas to black.

(* Create a canvas widget *) let canvas_id = X_ocaml_lib.X_canvas.create ~width:640 ~height:480;; (* Initialize WebGL when canvas is ready *) X_ocaml_lib.X_canvas.when_ready canvas_id (fun canvas -> (* Try to get WebGL context with alpha: false to force opaque background *) let opts = Jv.obj [| ("alpha", Jv.of_bool false) |] in let gl = Jv.call (X_ocaml_lib.X_canvas.get_offscreen canvas) "getContext" [| Jv.of_string "webgl"; opts |] in if Jv.is_null gl then print_endline "Unable to initialize WebGL. Your browser may not support it." else begin print_endline "WebGL context obtained successfully"; (* Set clear color to black, fully opaque *) let _ = Jv.call gl "clearColor" [| Jv.of_float 0.0; Jv.of_float 0.0; Jv.of_float 0.0; Jv.of_float 1.0 |] in print_endline "clearColor called with (0, 0, 0, 1)"; (* Get COLOR_BUFFER_BIT constant *) let color_buffer_bit = Jv.Int.get gl "COLOR_BUFFER_BIT" in Printf.printf "COLOR_BUFFER_BIT = %d\n" color_buffer_bit; (* Clear the color buffer *) let _ = Jv.call gl "clear" [| Jv.of_int color_buffer_bit |] in print_endline "gl.clear() called"; (* Check for WebGL errors *) let error = Jv.Int.get (Jv.call gl "getError" [||]) "valueOf" in if error = 0 then print_endline "No WebGL errors - rendering complete!" else Printf.printf "WebGL error: %d\n" error end )

Same as above, but clearing to a nice blue color instead of black.
